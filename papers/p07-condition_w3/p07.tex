\begin{abstract}
We prove Condition~W3: for every primitive Dirichlet character $\chi$ modulo
$q = 3^K$, the $L$-function $L(s, \chi)$ has no real zero in the region
$\sigma > 1 - c/\!\log q$, for an absolute, effectively computable constant
$c > 0$. The statement is classical, and its proof takes one line: the group
$(\Z/3^K\Z)^\times$ is cyclic of even order, so the only real non-principal
character modulo $3^K$ is the one induced from the character $\chi_{-3}$
modulo $3$; an exceptional zero can only belong to a real character; and
$L(\sigma,\chi_{-3}) > 0$ for $\sigma > 0$. As context we record, with
corrections, the arithmetic of $\Q(\omega)$ that the original version used:
the base change $\chi\circ\mathrm{N}$ of a character $\chi$ modulo $3^K$
takes the value $\chi(n)^2$ on the ideal $(n)$, its conductor is
$\mathfrak p^{2K-1}$ for $K \geq 2$, and
$L(s,\chi\circ\mathrm{N}) = L(s,\chi)L(s,\chi\chi_{-3})$. Condition~W3
does not make any level of distribution unconditional, and nothing in it is
specific to the prime $3$.
\end{abstract}

\subsection{Context}

This paper is p07 of a nine-paper programme~\cite{WakilCF}. The companion
paper~\cite{WakilEisenstein} (p04) concerns primes in arithmetic progressions
to the moduli $q = 3^K$. The original version of the present paper presented
Condition~W3 below as the third of three hypotheses ``(W1)--(W3)'' of Watt's
theorem~\cite{Watt1995}, and claimed that proving it made a level of
distribution $5/8$ for these moduli unconditional. Both claims are withdrawn
(Section~\ref{sec:relation}): Watt's theorem has no such hypotheses, and the
level $5/8$ for each single modulus $3^K$ is a conjecture
in~\cite{WakilEisenstein}, which would go beyond what the Generalised Riemann
Hypothesis gives. The name ``Condition~W3'' is kept as a label for the
following classical statement.

\subsection{Main theorem}

\begin{theorem}[Condition W3]
\label{thm:W3}
Let $K \geq 1$ and let $\chi$ be a primitive Dirichlet character modulo
$q = 3^K$. Then $L(s,\chi)$ has no real zero in the region
\[
  \sigma > 1 - \frac{c}{\log q}
\]
for an absolute, effectively computable constant $c > 0$.
\end{theorem}

The proof (Section~\ref{sec:proof}) uses only the cyclicity of
$(\Z/3^K\Z)^\times$, the classical zero-free region for Dirichlet
$L$-functions, and the positivity of $L(\sigma,\chi_{-3})$.
Sections~\ref{sec:field} and~\ref{sec:basechange} record, with corrections,
the arithmetic of $\Qomega$ that the original version used; it is context
only and is not needed for the proof. Section~\ref{sec:stark} explains why
the ``Stark--Baker theorem'' of the original version is withdrawn, and
Section~\ref{sec:relation} explains why W3 has no consequence for any level
of distribution beyond the Siegel--Walfisz range.

\subsection{Notation and conventions}

Throughout: $\omega = e^{2\pi i/3}$, so $\omega^2 + \omega + 1 = 0$.
We write $\K = \Qomega = \Q(\sqrt{-3})$, with ring of integers
$\OK = \Zomega$. The prime $\pp = (1-\omega)$ is the unique prime of
$\Zomega$ above $3$, with $\Nf(\pp) = 3$ and $(3) = \pp^2$. We write
$\varepsilon = \chi_{-3} = \bigl(\tfrac{-3}{\cdot}\bigr)$ for the
non-principal character modulo $3$, which is the quadratic character of
$\K/\Q$. For a Dirichlet character $\chi$, $\cond(\chi)$ is its conductor.
All constants are absolute and effectively computable unless stated
otherwise.

\section{Proof of Condition W3}
\label{sec:proof}
%% ─────────────────────────────────────────────────────────────────────────────

\begin{lemma}[Real characters modulo $3^K$]
\label{lem:real}
For $K \geq 1$, the group $(\Z/3^K\Z)^\times$ is cyclic of order
$2\cdot 3^{K-1}$, and there is exactly one real non-principal character
modulo $3^K$, namely the character induced by $\chi_{-3}$. Its $L$-function
is $L(s,\chi_{-3})$. In particular, for $K \geq 2$ every primitive character
modulo $3^K$ is complex.
\end{lemma}

\begin{proof}
The powers of $2$ modulo $9$ are $2, 4, 8, 7, 5, 1$, so $2$ is a primitive
root modulo $9$, hence modulo every $3^K$~\cite[Ch.~4]{IrelandRosen1990}
(and trivially modulo $3$). The character group of a cyclic group of even
order is cyclic of the same order and has exactly one element of order $2$.
The character induced by $\chi_{-3}$ is real and non-principal, so it is
that element. Since $\chi_{-3}(n) = 0$ already when $3 \mid n$, the induced
character coincides with $\chi_{-3}$ as a function on $\Z$, and the two
$L$-functions are equal. Its conductor is $3$, so it is not primitive
modulo $3^K$ for $K \geq 2$.
\end{proof}

\begin{lemma}[Positivity of $L(\sigma,\chi_{-3})$]
\label{lem:positive}
$L(\sigma,\chi_{-3}) > 0$ for every real $\sigma > 0$.
\end{lemma}

\begin{proof}
For $\sigma > 0$ the series $\sum_n \chi_{-3}(n)n^{-\sigma}$ converges, and
grouping consecutive non-zero terms gives
\[
  L(\sigma,\chi_{-3}) = \sum_{m \geq 0}\bigl((3m+1)^{-\sigma} - (3m+2)^{-\sigma}\bigr),
\]
a series of positive terms.
\end{proof}

For example, $L(0,\chi_{-3}) = 1/3$ and
$L(1,\chi_{-3}) = \pi/(3\sqrt3) = 0.6045\ldots$.

\begin{theorem}[Classical zero-free region; see {\cite[Ch.~14]{Davenport2000}}]
\label{thm:classical}
There is an absolute, effectively computable constant $c_0 \in (0,1]$ with
the following property. Let $q \geq 3$ and let $\chi$ be a non-principal
character modulo $q$. If $\chi$ is not real, $L(s,\chi)$ has no zero in the
region $\sigma \geq 1 - c_0/\log\bigl(q(|t|+2)\bigr)$. If $\chi$ is real,
$L(s,\chi)$ has at most one zero in this region, and such a zero is real
and simple.
\end{theorem}

\begin{proof}[Proof of Theorem~\ref{thm:W3}]
Let $c = c_0/2$, with $c_0$ as in Theorem~\ref{thm:classical}, and suppose
that $\beta$ is a real zero of $L(s,\chi)$ with $\beta > 1 - c/\log q$. Since
$L(s,\chi) \neq 0$ for $\sigma \geq 1$, we have $\beta < 1$. Since $q \geq 3$,
$\log(2q) \leq 2\log q$, so $\beta > 1 - c_0/\log(2q)$, and $s = \beta$
lies in the region of Theorem~\ref{thm:classical} (with $t = 0$).

If $K \geq 2$, then $\chi$ is not real (Lemma~\ref{lem:real}), and
Theorem~\ref{thm:classical} excludes $\beta$. If $K = 1$, the only primitive
character modulo $3$ is $\chi = \chi_{-3}$, and $L(\beta,\chi_{-3}) > 0$ by
Lemma~\ref{lem:positive}. In either case we reach a contradiction.
\end{proof}

\begin{remark}[The one-line argument, and a slightly stronger statement]
\label{rem:oneline}
In one line: $(\Z/3^K\Z)^\times$ is cyclic of even order, so the only real
non-principal character modulo $3^K$ is induced from $\chi_{-3}$; an
exceptional zero can only belong to a real character; so any exceptional
zero modulo $3^K$ would be a zero of the single fixed function
$L(s,\chi_{-3})$, which is positive on $(0,1]$. The same argument shows that
no non-principal character modulo $3^K$, primitive or not, has a zero in the
region of Theorem~\ref{thm:classical} with $q = 3^K$: the moduli $3^K$ have
no exceptional zero. Consequently the prime number theorem for arithmetic
progressions to these moduli holds in the Siegel--Walfisz range with
effectively computable constants~\cite[Ch.~20]{Davenport2000}.
\end{remark}

\begin{remark}[Nothing here is specific to $3$]
\label{rem:notspecific}
The proof uses only that $(\Z/3^K\Z)^\times$ is cyclic of even order and
that one fixed real $L$-function has no real zero near $1$. For every odd
prime $\ell$, $(\Z/\ell^K\Z)^\times$ is cyclic of even order, and the only
real non-principal character modulo $\ell^K$ is induced from the Legendre
symbol $(\cdot/\ell)$. Hence, for each fixed $\ell$, the same argument proves
Condition~W3 for the moduli $\ell^K$, with a constant $c_\ell > 0$ that
depends on $\ell$ (through the largest real zero of $L(s,(\cdot/\ell))$ in
$(0,1)$, if there is one). A constant uniform in $\ell$ would exclude Siegel
zeros for the real characters of prime conductor, which is an open problem.
Likewise, the field $\Qomega$ plays no special role: $\Q(\sqrt{-7})$,
$\Q(\sqrt{-11})$, $\Q(\sqrt{-19})$, $\Q(\sqrt{-43})$, $\Q(\sqrt{-67})$ and
$\Q(\sqrt{-163})$ also have class number~$1$, and each is ramified at a
single prime~$\ell$, with $(\ell) = \mathfrak l^2$.
\end{remark}

\section{The field $\Qomega$ (context)}
\label{sec:field}
%% ─────────────────────────────────────────────────────────────────────────────

\begin{proposition}[Arithmetic of $\Qomega$]
\label{prop:CM}
The field $\K = \Qomega = \Q(\sqrt{-3})$ is imaginary quadratic, hence a CM
field; its maximal totally real subfield is $\Q$. Its ring of integers is
$\Zomega$, its discriminant is $-3$, its class number is $h_\K = 1$, and its
unit group is $\mu_6 = \{\pm1, \pm\omega, \pm\omega^2\}$.
\end{proposition}

\begin{proof}
Since $-3 \equiv 1 \pmod 4$, the ring of integers is
$\Z\bigl[\tfrac{1+\sqrt{-3}}{2}\bigr] = \Zomega$, with discriminant $-3$.
The Minkowski bound equals $(2/\pi)\sqrt{3} \approx 1.10$, so every ideal
class contains an integral ideal of norm at most $1.10$, that is, of norm
$1$; hence $h_\K = 1$. The units are the six solutions of
$a^2 - ab + b^2 = 1$, namely $\pm1, \pm\omega, \pm\omega^2$.
\end{proof}

\begin{remark}
$\K$ is one of the nine imaginary quadratic fields of class number one,
$\Q(\sqrt{-d})$ with $d \in \{1,2,3,7,11,19,43,67,163\}$, as proved by
Heegner~\cite{Heegner1952}, Baker~\cite{Baker1966} and
Stark~\cite{Stark1967}. That $\Zomega$ is a principal ideal domain was known
long before; the class-number-one theorem plays no role in this paper.
\end{remark}

\begin{proposition}[Ramification of $3$]
\label{prop:ram}
In $\Zomega$, the rational prime $3$ ramifies: $(3) = \pp^2$ where
$\pp = (1-\omega)$ and $\Nf(\pp) = 3$.
\end{proposition}

\begin{proof}
$\Nf(1-\omega) = (1-\omega)(1-\bar\omega) = 1 - (\omega + \bar\omega) +
\omega\bar\omega = 1 - (-1) + 1 = 3$; equivalently, $\Nf(a+b\omega) =
a^2-ab+b^2$ gives $\Nf(1-\omega) = 1+1+1 = 3$. Since $1+\omega+\omega^2=0$,
we have $3 = -\omega^2(1-\omega)^2$, and $-\omega^2 \in \Zomega^\times$.
\end{proof}

\begin{remark}
The prime $3$ is the only rational prime that ramifies in $\K$: a prime
$\ell$ ramifies in $\K$ if and only if $\ell \mid \disc(\K) =
-3$~\cite{Neukirch1999}. This is not special to $3$: in the same way, $\ell$
is the only prime ramified in $\Q(\sqrt{-\ell})$ for
$\ell = 7, 11, 19, 43, 67, 163$ (Remark~\ref{rem:notspecific}).
\end{remark}

\begin{lemma}[Splitting of primes $\ell \neq 3$]
\label{lem:split}
A prime $\ell \neq 3$ splits in $\K$ into two distinct primes of norm $\ell$
if $\ell \equiv 1 \pmod 3$, and is inert, with $\Zomega/(\ell) \cong
\F_{\ell^2}$, if $\ell \equiv 2 \pmod 3$. In particular $2$ is inert and
$\Zomega/(2) \cong \F_4$.
\end{lemma}

\begin{proof}
Since $\Zomega \cong \Z[x]/(x^2+x+1)$, the prime $\ell$ splits if
$x^2 + x + 1$ has a root modulo $\ell$ and is inert otherwise (the
Dedekind--Kummer criterion~\cite{Neukirch1999}; for $\ell \neq 3$ there is no
repeated root, as the discriminant of $x^2+x+1$ is $-3$). A root $r$
satisfies $r^3 = 1$ and $r \neq 1$ (as $3 \not\equiv 0 \pmod \ell$), so a
root exists if and only if $\F_\ell^\times$ has an element of order $3$,
that is, if and only if $3 \mid \ell - 1$. For $\ell = 2$ this agrees with
the usual criterion for $2$ in $\Q(\sqrt d)$ with $d \equiv 1 \pmod 4$
($2$ splits if $d \equiv 1 \pmod 8$ and is inert if $d \equiv 5 \pmod 8$),
since $-3 \equiv 5 \pmod 8$.
\end{proof}

\section{Base change of characters modulo $3^K$ (context)}
\label{sec:basechange}
%% ─────────────────────────────────────────────────────────────────────────────

\subsection{Hecke characters of finite order}

Let $\mathfrak m$ be a nonzero ideal of $\OK$. Since $\K$ is imaginary
quadratic, a Hecke character of finite order of $\K$ with modulus
$\mathfrak m$ is a homomorphism $\Psi$ from the group of fractional ideals
of $\K$ prime to $\mathfrak m$ to $\C^\times$ such that $\Psi((\alpha)) = 1$
for every $\alpha \in \OK$ with $\alpha \equiv 1 \pmod{\mathfrak m}$; that
is, a character of the ray class group modulo $\mathfrak m$. Its conductor
$\cond(\Psi)$ is the smallest such modulus. We put $\Psi(\mathfrak a) = 0$
for integral ideals $\mathfrak a$ not prime to $\mathfrak m$, and
\[
  L(s, \Psi) = \sum_{\mathfrak a} \Psi(\mathfrak a)\Nf(\mathfrak a)^{-s}
  = \prod_{\qq} \bigl(1 - \Psi(\qq)\,\Nf(\qq)^{-s}\bigr)^{-1},
  \qquad \Re s > 1,
\]
where $\mathfrak a$ runs over nonzero integral ideals and $\qq$ over prime
ideals.

\subsection{The base change $\chi\circ\Nf$}

\begin{proposition}[Base change of characters modulo $3^K$]
\label{prop:lift}
Let $K \geq 1$, let $\chi$ be a Dirichlet character modulo $3^K$, and put
$\Psi_\chi(\mathfrak a) = \chi(\Nf\mathfrak a)$ for nonzero integral ideals
$\mathfrak a$ of $\Zomega$ (so $\Psi_\chi(\mathfrak a) = 0$ if
$\pp \mid \mathfrak a$).
\begin{enumerate}[label=(\roman*)]
  \item $\Psi_\chi$ is a Hecke character of finite order with modulus
    $\pp^{2K}$, and $\Psi_\chi((n)) = \chi(n)^2$ for every $n \in \Z$.
  \item The image of the norm map $(\OK/\pp^{2K})^\times \to
    (\Z/3^K\Z)^\times$ is the subgroup $\{n \equiv 1 \pmod 3\}$, of
    index~$2$. Consequently $\Psi_\chi = \Psi_{\chi\varepsilon}$, and the map
    $\chi \mapsto \Psi_\chi$ is two-to-one.
  \item If $\chi$ is primitive, then $\Psi_\chi$ is the trivial character
    when $K = 1$, and $\cond(\Psi_\chi) = \pp^{2K-1}$ when $K \geq 2$.
\end{enumerate}
\end{proposition}

\begin{proof}
(i) The norm is multiplicative, so $\Psi_\chi$ is multiplicative, and since
units have norm $1$, $\Psi_\chi((\alpha)) = \chi(\Nf\alpha)$ does not depend
on the generator $\alpha$. If $\alpha \in \OK$ and $\alpha \equiv 1
\pmod{\pp^{2K}}$, then $\alpha = 1 + 3^K\beta$ with $\beta \in \OK$, and
$\Nf(\alpha) = 1 + 3^K\operatorname{Tr}(\beta) + 3^{2K}\Nf(\beta) \equiv 1
\pmod{3^K}$, so $\Psi_\chi((\alpha)) = 1$. Finally $\Nf((n)) = n^2$.

(ii) For $\alpha = a + b\omega$ we have $\Nf(\alpha) = a^2 - ab + b^2 \equiv
(a+b)^2 \pmod 3$, so every norm prime to $3$ is $\equiv 1 \pmod 3$.
Conversely, the squares in the cyclic group $(\Z/3^K\Z)^\times$ of order
$2\cdot 3^{K-1}$ form its unique subgroup of index~$2$, which is therefore
$\{n \equiv 1 \pmod 3\}$, and $m^2 = \Nf(m)$ for $m \in \Z$. Hence the image
is exactly $\{n \equiv 1 \pmod 3\}$. Two characters modulo $3^K$ agree on
this subgroup if and only if they are equal or differ by the unique
non-trivial character trivial on it, which is $\varepsilon$; this gives the
last two assertions.

(iii) For $K = 1$, $\chi = \varepsilon$, and $\varepsilon(\Nf\mathfrak a) =
1$ for every $\mathfrak a$ prime to $\pp$, by (ii). Let $K \geq 2$. One
checks that $\operatorname{Tr}(\OK) = \Z$ and $\operatorname{Tr}(\pp) =
3\Z$ (indeed $\operatorname{Tr}((1-\omega)(a+b\omega)) = 3a$); since
$\pp^{2j} = 3^j\OK$, it follows that $\operatorname{Tr}(\pp^m) =
3^{\lceil m/2\rceil}\Z$. If $\alpha = 1 + \beta$ with $\beta \in
\pp^{2K-1}$, then $\Nf(\alpha) = 1 + \operatorname{Tr}(\beta) + \Nf(\beta)
\equiv 1 \pmod{3^K}$, because $\operatorname{Tr}(\beta) \in 3^K\Z$ and
$\Nf(\beta) \in 3^{2K-1}\Z$. So $\pp^{2K-1}$ is a modulus for $\Psi_\chi$.
On the other hand, for $t \in \Z$ the element $\alpha = 1 + 3^{K-1}t$ is
$\equiv 1 \pmod{\pp^{2K-2}}$, and $\Nf(\alpha) = (1+3^{K-1}t)^2 \equiv
1 + 2\cdot3^{K-1}t \pmod{3^K}$ because $2K - 2 \geq K$; these values run
through all classes $\equiv 1 \pmod{3^{K-1}}$. A primitive $\chi$ is
non-trivial on that subgroup, so $\pp^{2K-2}$ is not a modulus for
$\Psi_\chi$, and $\cond(\Psi_\chi) = \pp^{2K-1}$.
\end{proof}

The conductor in (iii) agrees with the conductor--discriminant formula
$|\disc(\K)|\cdot\Nf\cond(\Psi_\chi) = \cond(\chi)\,\cond(\chi\varepsilon)$,
which for $K \geq 2$ reads $3 \cdot 3^{2K-1} = 3^K\cdot 3^K$.

\begin{remark}[The lifting identity of the original version]
\label{rem:nolift}
The original version (former Proposition~2.5) claimed, for every primitive
$\chi$ modulo $3^K$, a primitive Hecke character $\Psi_\chi$ of conductor
$\pp^{2K}$ with $\chi(n) = \Psi_\chi((n))$ for all $n$ prime to $3$. No
such character exists when $\chi$ is odd: since $(n) = (-n)$, any function
of ideals takes the same value at $n$ and $-n$, which forces
$\chi(-1) = 1$. Half of the characters modulo $3^K$ are odd, because $-1$ is
the element of order $2$ of the cyclic group $(\Z/3^K\Z)^\times$. The
construction actually given was $\Psi_\chi = \chi\circ\Nf$, for which (i)
gives $\chi(n)^2$; its conductor is $\pp^{2K-1}$, not $\pp^{2K}$, by (iii);
and the norm map is not surjective, by (ii). In particular the ramification
of $3$ does not make conductors ``match''. The opposite is true: for a prime
$\ell \neq 3$, which is unramified, the base change of a primitive character
modulo $\ell^K$ has conductor exactly $\ell^K\OK$, whereas at the ramified
prime $3$ the conductor drops to $\pp^{2K-1}$.
\end{remark}

\subsection{The $L$-function factorisation}

\begin{lemma}[$L$-function factorisation]
\label{lem:factor}
For every Dirichlet character $\chi$ modulo $3^K$, $K \geq 1$,
\[
  L(s, \Psi_\chi) = L(s, \chi) \cdot L(s, \chi\varepsilon) \qquad (\Re s > 1),
\]
and hence for all $s$, by analytic continuation.
\end{lemma}

\begin{proof}
Both sides are Euler products; we compare the local factors at each rational
prime $\ell$.

\emph{$\ell = 3$.} $\Psi_\chi(\pp) = \chi(3) = 0$, and $\chi(3) =
\chi\varepsilon(3) = 0$, so both local factors are $1$.

\emph{$\ell \equiv 1 \pmod{3}$ (split).} Write $\ell\Zomega = \qq_1\qq_2$
with $\qq_1 \neq \qq_2$ of norm $\ell$. The local factor of
$L(s, \Psi_\chi)$ is $(1-\chi(\ell)\ell^{-s})^{-2}$. Since
$\varepsilon(\ell) = 1$, this equals the product of the local factors of
$L(s,\chi)$ and $L(s,\chi\varepsilon)$.

\emph{$\ell \equiv 2 \pmod{3}$ (inert, including $\ell = 2$).} Then
$\ell\Zomega = \qq$ is a prime of norm $\ell^2$, and $\Psi_\chi(\qq) =
\chi(\ell^2) = \chi(\ell)^2$. Since $\varepsilon(\ell) = -1$, the product of
the local factors of $L(s,\chi)$ and $L(s,\chi\varepsilon)$ is
$(1-\chi(\ell)\ell^{-s})^{-1}(1+\chi(\ell)\ell^{-s})^{-1}
= (1-\chi(\ell)^2\ell^{-2s})^{-1}$, the local factor of $L(s,\Psi_\chi)$.
\end{proof}

\begin{remark}[The twist $\chi\varepsilon$, and the case $K = 1$]
\label{rem:chieps}
For $K \geq 2$ and $\chi$ primitive, $\chi\varepsilon$ is again primitive
modulo $3^K$, since $\varepsilon$ is trivial on $\{n \equiv 1 \pmod{3^{K-1}}\}$,
and $\Psi_{\chi\varepsilon} = \Psi_\chi$. For $K = 1$, $\chi = \varepsilon$,
and $\chi\varepsilon = \varepsilon^2$ is the principal character modulo $3$,
of conductor $1$. Then $\Psi_\varepsilon$ is trivial, and
Lemma~\ref{lem:factor} reads $\zeta_\K(s)(1 - 3^{-s}) =
L(s,\varepsilon)\,\zeta(s)(1-3^{-s})$, that is, $\zeta_\K(s) =
\zeta(s)L(s,\varepsilon)$. The original version stated that $\chi\varepsilon$
is non-principal of conductor $3^K$ for all $K \geq 1$, and that
$\Psi_\varepsilon$ is primitive of conductor $\pp^2$; both are false for
$K = 1$. Indeed the ray class group of $\K$ modulo $\pp^2$ is trivial, since
the six units map onto $(\OK/\pp^2)^\times$, which has order $6$; so no
non-trivial character of conductor $\pp^2$ exists.
\end{remark}

\begin{remark}
Lemma~\ref{lem:factor} expresses $L(s,\Psi_\chi)$ through two Dirichlet
$L$-functions modulo $3^K$. A zero-free region for $L(s,\Psi_\chi)$ is
therefore the same statement as a zero-free region for the pair $L(s,\chi)$,
$L(s,\chi\varepsilon)$, and the passage to $\Qomega$ adds nothing to
Section~\ref{sec:proof}.
\end{remark}

\section{The former ``Stark--Baker theorem'' (withdrawn)}
\label{sec:stark}
%% ─────────────────────────────────────────────────────────────────────────────

The original version derived Theorem~\ref{thm:W3} from the following
statement, presented as a ``Stark--Baker theorem'' (former Theorem~3.2):
\emph{for every CM field $F$ and every Hecke character $\Psi$ of $F$ of
conductor $\mathfrak f$, $L(s,\Psi)$ has no real zero in
$\sigma > 1 - c_F/\log(\Nf(\mathfrak f)\,|\disc(F)|)$, with $c_F > 0$
effective and depending only on $[F:\Q]$.} This is not a known theorem, and
it is withdrawn, together with its proof sketch and the former Remark~3.3
and Corollary~3.4.

As stated it would solve the Landau--Siegel zero problem. Take
$F = \Q(\sqrt{-d})$, any imaginary quadratic field (hence a CM field of
degree~$2$), and $\Psi$ trivial. Then $L(s,\Psi) = \zeta_F(s) =
\zeta(s)L(s,\chi_{d_F})$, where $d_F$ is the discriminant of $F$ and
$\chi_{d_F}$ the associated quadratic character, and $\zeta(\sigma) < 0$ for
$0 < \sigma < 1$. The statement would therefore give that $L(s,\chi_{d_F})$
has no real zero in $\sigma > 1 - c/\log|d_F|$ for every imaginary quadratic
field, with $c$ absolute and effective. That is the open Landau--Siegel
problem for these characters; via the class number formula it would also
give an effective lower bound $h(d_F) \gg |d_F|^{1/2}/\log|d_F|$, far
beyond the known effective lower bounds.

What Heegner~\cite{Heegner1952}, Baker~\cite{Baker1966} and
Stark~\cite{Stark1967} proved is the class-number-one theorem: the nine
fields listed after Proposition~\ref{prop:CM} are the only imaginary
quadratic fields of class number~$1$. It gives no zero-free region for
Hecke $L$-functions of $\Qomega$. The remark that Siegel zeros ``remain
possible'' only for non-CM fields is also withdrawn, since every imaginary
quadratic field is a CM field. None of this affects Theorem~\ref{thm:W3},
whose proof in Section~\ref{sec:proof} uses only the cyclicity of
$(\Z/3^K\Z)^\times$.

\section{Relation to the companion paper and to Watt's theorem}
\label{sec:relation}
%% ─────────────────────────────────────────────────────────────────────────────

The original Sections~5--7 (a proposition that the argument is specific to
$3$, a ``coherence'' proposition, a table verifying ``Watt's conditions''
W1--W3, and a corollary asserting that the level of distribution
$\theta_W = 5/8$ for cascade moduli holds unconditionally) are withdrawn,
for the following reasons.

\begin{enumerate}[label=(\arabic*)]
  \item \emph{Watt's theorem has no conditions W1--W3.} N.~Watt~\cite{Watt1995}
    proves a mean-value estimate in the $t$-aspect for
    $\int_0^T|\zeta(\tfrac12+it)|^4\,\bigl|\sum_{m\leq M}a_m m^{-it}\bigr|^2\,dt$.
    It involves no family of Dirichlet characters modulo $q$ and has no
    hypotheses ``$M \leq q$'', ``$MR \leq q^2$'' or ``no Siegel zeros''.
    Moreover, with the parameters $M = Q^{2/3}$ and $R = Q^{1/3}$ used in the
    original version of~\cite{WakilEisenstein}, ``W1'' fails for every
    $3^K < Q^{2/3}$ and ``W2'' for every $3^K < Q^{1/2}$, so these were not
    verified for the family either (former Theorem~7.1).
  \item \emph{W3 is unrelated to $\theta \leq 7/32$.} The bound
    $\theta \leq 7/32$ is the Kim--Sarnak bound~\cite{KimSarnak2003}
    $\lambda_1 \geq 1/4 - (7/64)^2$ towards Selberg's eigenvalue conjecture,
    written in the normalisation $\theta = \sqrt{1 - 4\lambda_1}$. It is an
    unconditional theorem about Maass forms of every level, and neither the
    absence of Siegel zeros nor any zero-density estimate implies it. The
    derivation of it in the original version is withdrawn, together with
    the zero-density estimate quoted there, whose exponent and source we
    could not confirm.
  \item \emph{W3 makes nothing unconditional.} Theorem~1.1
    of~\cite{WakilEisenstein}, a sum over the moduli $3^K \leq x^\theta$,
    holds for every fixed $\theta < 1$ by the trivial bound and the
    Siegel--Walfisz theorem. It does not need W3, although
    Remark~\ref{rem:oneline} makes its implied constant effective. The
    per-modulus statement for $3^K$ up to $x^{5/8-\varepsilon}$ is a conjecture
    in~\cite{WakilEisenstein}, which would go beyond the Generalised Riemann
    Hypothesis. A zero-free region of width $c/\log q$ cannot help with it:
    for $q = x^\theta$ one has $x^{1 - c/\log q} = e^{-c/\theta}x$, so such a
    region gives no saving for moduli of size a power of $x$. The former
    Corollary~7.2 is therefore withdrawn. (Its reference to an annotation
    ``[NOTE]'' in~\cite{WakilEisenstein} was also wrong: there is none.)
  \item \emph{No ``coherence''.} The former Proposition~6.1 asserted that one
    fact, the ramification $(3) = \pp^2$, drives three steps
    of~\cite{WakilEisenstein}: a three-level filtration, Kloosterman
    cancellation beyond the Weil bound, and a conductor match. All three are
    false. The $(1-\omega)$-adic filtration of $\Zomega/(3^K)$ has $2K$
    steps, not three. The bound $|S(a,b;3^K)| \ll (3^K)^{3/8+\varepsilon}$
    fails, since $S(1,1;3^K) = 2\cdot 3^{K/2}\cos(4\pi/3^K)$ for even
    $K \geq 2$ (see~\cite{WakilEisenstein}); and cyclicity of
    $(\Z/\ell^K\Z)^\times$ holds for every odd prime $\ell$ and has nothing
    to do with $\Zomega$. The conductor does not match
    (Remark~\ref{rem:nolift}).
  \item \emph{Not specific to $3$.} The former Proposition~5.1 is withdrawn;
    see Remark~\ref{rem:notspecific}. Its proof also stated that $2$ splits
    in $\K$ ``since $-3 \equiv 1 \pmod 8$''. In fact $-3 \equiv 5 \pmod 8$,
    and $2$ is inert (Lemma~\ref{lem:split}). Its third property, that
    characters of $(\Z/3^K\Z)^\times$ correspond precisely to Hecke
    characters of $\K$ of conductor $\pp^{2K}$, is false as well: the map
    $\chi \mapsto \chi\circ\Nf$ is two-to-one
    (Proposition~\ref{prop:lift}(ii)), and the ray class group of $\K$
    modulo $\pp^{2K}$ has order $3^{2K-2}$, not $2\cdot 3^{K-1}$.
\end{enumerate}
